READ # Reads user input
JUMP.EQ.0 L1 # Jumps to the EVEN label if the input is 0

LOOP: # The LOOP label
PUSH 2 # Pushes 2 to the stack
SUB # Subtracts 2 from second top-most item from the stack (in this case the user input)
JUMP.EQ.0 EVEN # Jumps to the EVEN label if the result is 0
JUMP.GT.0 LOOP # Jumps to the LOOP label if the result is still greater than 0, this repeats until the results is 0, which will make it jump to EVEN, or until it turns negative, which makes it skip this jump statement
PRINT "odd" # Prints "odd" in the output/console
HALT # Stops the program

EVEN: # The EVEN label
PRINT "even" # Prints "even" in the output/console
HALT # Stops the program
